import { Platform, Linking, Alert } from 'react-native';
import * as FileSystem from 'expo-file-system';
import * as Permissions from 'expo-permissions';

const downloadApk = async (apkUrl) => {
  try {
    if (Platform.OS === 'android') {
      const { status } = await Permissions.askAsync(Permissions.MEDIA_LIBRARY);
      if (status !== 'granted') {
        Alert.alert('Permiso denegado', 'Necesitas conceder acceso al almacenamiento para descargar la APK.');
        return;
      }

      // Obtén la ruta donde guardar la APK
      const fileUri = FileSystem.documentDirectory + 'app_name.apk';

      // Descarga el archivo APK desde la URL
      const downloadResumable = FileSystem.createDownloadResumable(
        apkUrl,
        fileUri
      );
      const { uri } = await downloadResumable.downloadAsync();

      // Aquí lanzas el Intent para instalar la APK
      const apkUri = uri;
      Linking.openURL(apkUri); // O puedes usar un paquete como react-native-fs para moverlo a una ubicación más adecuada

    } else {
      Alert.alert('Solo Android', 'Esta funcionalidad solo está disponible en Android.');
    }
  } catch (error) {
    console.error('Error al descargar APK:', error);
  }
};
